# RAG Chatbot Python Script

I used Claude 4.5 Sonnet to generate the chatbot Python script used in this project.

The script evolved gradually over time. Initially my intention was to generate just a simple, bare-bones script, but as the project proceeded, I realised I needed extra capabilities—logging, configurability, batch mode—so the script expanded and became quite complex. It also became somewhat untidy because the AI sometimes overdelivered on what I was asking for, such as the large number of configuration options that I didn’t really need. However, I did not feel it was worth my time to address this as it was not central to what I was trying to achieve. The chatbot script is for testing purposes only and was never intended to be a user-friendly utility.

The script has quite a few dependencies, and the LLM runs locally, so it takes some effort to get this running, and then a typical response time for any query (on my laptop) is about 2-3 minutes.

class DocumentFormat(Enum):
    """Supported document formats."""
    PLAIN_TEXT = "txt"
    PDF = "pdf"
    HTML = "html"
    DITA = "dita"
    
    @classmethod
    def from_extension(cls, extension: str) -> Optional['DocumentFormat']:
        extension = extension.lstrip('.').lower()
        extension_map = {
            'txt': cls.PLAIN_TEXT, 'text': cls.PLAIN_TEXT,
            'pdf': cls.PDF,
            'html': cls.HTML, 'htm': cls.HTML,
            'dita': cls.DITA, 'ditamap': cls.DITA,
        }
        return extension_map.get(extension)
    
    def get_extensions(self) -> List[str]:
        if self == DocumentFormat.PLAIN_TEXT: return ['.txt', '.text']
        elif self == DocumentFormat.PDF: return ['.pdf']
        elif self == DocumentFormat.HTML: return ['.html', '.htm']
        elif self == DocumentFormat.DITA: return ['.dita', '.ditamap']
        return []
    
    def get_description(self) -> str:
        descriptions = {
            DocumentFormat.PLAIN_TEXT: "Plain text files",
            DocumentFormat.PDF: "PDF documents",
            DocumentFormat.HTML: "HTML web pages",
            DocumentFormat.DITA: "DITA XML topics"
        }
        return descriptions.get(self, "Unknown format")
    
    def is_available(self) -> bool:
        if self == DocumentFormat.PLAIN_TEXT: return True
        elif self == DocumentFormat.PDF: return PDF_SUPPORT
        elif self == DocumentFormat.HTML: return HTML_SUPPORT
        elif self == DocumentFormat.DITA: return DITA_SUPPORT
        return False
    
    def __str__(self):
        return self.value.upper()

class SessionLogger:

    def log_query(self, question: str, answer: str, response_time: float, 
                sources: List[Document] = None, from_cache: bool = False):
        self.query_count += 1
        timestamp = datetime.now()
        
        source_info = []
        source_chunks = []  # ADD THIS LINE
        format_distribution = {}
        
        if sources:
            for doc in sources[:5]:
                filename = doc.metadata.get('filename', 'Unknown')
                fmt = doc.metadata.get('format', 'unknown')
                page = doc.metadata.get('page_number')
                
                format_distribution[fmt] = format_distribution.get(fmt, 0) + 1
                
                source_str = f"{filename} [{fmt}]"
                if page:
                    source_str += f" (page {page})"
                source_info.append(source_str)
                
                # ADD THESE LINES - Store chunk content
                source_chunks.append({
                    'filename': filename,
                    'format': fmt,
                    'page': page,
                    'chunk_text': doc.page_content
                })
        
        query_record = {
            'query_number': self.query_count,
            'timestamp': timestamp.isoformat(),
            'question': question,
            'answer': answer,
            'response_time': response_time,
            'from_cache': from_cache,
            'sources': source_info,
            'source_chunks': source_chunks,  # ADD THIS LINE
            'source_count': len(sources) if sources else 0,
            'format_distribution': format_distribution
        }
        
        self.queries.append(query_record)
        self._write_text_log(query_record, timestamp)
        self._write_json_log()
    
    def _write_text_log(self, record: Dict, timestamp: datetime):
        dash_separator = "-" * 80
        chunk_separator = "." * 80  # ADD THIS LINE
        
        with open(self.text_log_path, 'a', encoding='utf-8') as f:
            f.write(dash_separator + "\n")
            f.write(f"[Q{record['query_number']}] {timestamp.strftime('%H:%M:%S')} | User asked:\n")
            f.write(f"{record['question']}\n\n")
            
            cache_note = " [FROM CACHE]" if record['from_cache'] else ""
            f.write(f"[A{record['query_number']}] {timestamp.strftime('%H:%M:%S')} | ")
            f.write(f"Bot answered ({record['response_time']:.2f}s){cache_note}:\n")
            f.write(f"{record['answer']}\n\n")
            
            if record['sources']:
                f.write(f"Sources used ({record['source_count']}):\n")
                for source in record['sources']:
                    f.write(f"  - {source}\n")
                if record['format_distribution']:
                    f.write("\nFormat distribution: ")
                    f.write(", ".join(f"{fmt}: {count}" for fmt, count in record['format_distribution'].items()))
                    f.write("\n")
                
                # ADD THIS SECTION - Write chunk content
                if record.get('source_chunks'):
                    f.write("\n" + chunk_separator + "\n")
                    f.write("CHUNK CONTENT USED FOR THIS ANSWER:\n")
                    f.write(chunk_separator + "\n\n")
                    
                    for i, chunk_info in enumerate(record['source_chunks'], 1):
                        f.write(f"[CHUNK {i}] {chunk_info['filename']} [{chunk_info['format'].upper()}]")
                        if chunk_info.get('page'):
                            f.write(f" (page {chunk_info['page']})")
                        f.write("\n")
                        f.write("-" * 80 + "\n")
                        f.write(chunk_info['chunk_text'])
                        f.write("\n" + "-" * 80 + "\n\n")
                    
                    f.write(chunk_separator + "\n")
                
                f.write("\n")
            f.write("\n")
    
    def _write_json_log(self):
        session_data = {
            'session_id': self.session_id,
            'started': self.session_start.isoformat(),
            'ended': datetime.now().isoformat(),
            'system': self.system_info,
            'configuration': {
                'documents_folder': self.config.documents_folder,
                'selected_formats': [fmt.value for fmt in self.config.selected_formats],
                'chunk_size': self.config.chunk_size,
                'chunk_overlap': self.config.chunk_overlap,
                'embedding_model': self.config.embedding_model,
                'llm_model_path': self.config.model_path,
            },
            'document_statistics': self.document_stats,
            'total_queries': self.query_count,
            'queries': self.queries,
            'statistics': self._get_statistics()
        }
        
        with open(self.json_log_path, 'w', encoding='utf-8') as f:
            json.dump(session_data, f, indent=2, ensure_ascii=False)
    
    def _get_statistics(self) -> Dict:
        if not self.queries:
            return {}
        
        response_times = [q['response_time'] for q in self.queries]
        cache_hits = sum(1 for q in self.queries if q['from_cache'])
        
        all_formats = {}
        for q in self.queries:
            for fmt, count in q.get('format_distribution', {}).items():
                all_formats[fmt] = all_formats.get(fmt, 0) + count
        
        return {
            'total_queries': len(self.queries),
            'avg_response_time': sum(response_times) / len(response_times),
            'min_response_time': min(response_times),
            'max_response_time': max(response_times),
            'cache_hits': cache_hits,
            'cache_misses': len(self.queries) - cache_hits,
            'avg_sources_used': sum(q['source_count'] for q in self.queries) / len(self.queries),
            'total_sources_used': sum(q['source_count'] for q in self.queries),
            'format_usage': all_formats
        }
    
    def finalize(self):
        session_end = datetime.now()
        duration = session_end - self.session_start
        stats = self._get_statistics()
        
        separator = "=" * 80
        
        with open(self.text_log_path, 'a', encoding='utf-8') as f:
            f.write(separator + "\n")
            f.write("SESSION SUMMARY\n")
            f.write(separator + "\n")
            f.write(f"Session ended:        {session_end.strftime('%Y-%m-%d %H:%M:%S')}\n")
            f.write(f"Duration:             {duration}\n")
            f.write(f"Total queries:        {stats.get('total_queries', 0)}\n")
            
            if stats.get('total_queries', 0) > 0:
                f.write("\nPerformance:\n")
                f.write(f"  Avg response time:  {stats['avg_response_time']:.2f}s\n")
                f.write(f"  Fastest response:   {stats['min_response_time']:.2f}s\n")
                f.write(f"  Slowest response:   {stats['max_response_time']:.2f}s\n")
            
            f.write(separator + "\n")
        
        self._write_json_log()
    
    def get_log_path(self) -> str:
        return str(self.text_log_path)
    
    def get_json_path(self) -> str:
        return str(self.json_log_path)

class BatchQueryRunner:
    """Run multiple queries in batch mode."""
    
    def __init__(self, chatbot: 'InteractiveChatbot'):
        self.chatbot = chatbot
        self.results = []
    
    def run_from_file(self, questions_file: str, output_file: str = None):
        """Run batch queries from file."""
        separator = "=" * 70
        dash_separator = "-" * 70
        
        print()
        print(separator)
        print("🚀 BATCH QUERY MODE")
        print(separator)
        print(f"Loading questions from: {questions_file}\n")
        
        questions = self._load_questions(questions_file)
        
        if not questions:
            print("❌ No questions found!")
            return
        
        print(f"📝 Found {len(questions)} question(s)\n")
        
        self.results = []
        total_start = time.time()
        
        for i, question in enumerate(questions, 1):
            print(dash_separator)
            print(f"[Q{i}/{len(questions)}] {question['text']}")
            
            if question.get('category'):
                print(f"Category: {question['category']}")
            print()
            
            start_time = time.time()
            response = self.chatbot.query(question['text'])
            elapsed = time.time() - start_time
            
            if "error" in response:
                print(f"❌ Error: {response['error']}\n")
                result = {
                    'question': question['text'],
                    'category': question.get('category', 'uncategorized'),
                    'error': response['error'],
                    'response_time': elapsed
                }
            else:
                print("🤖 Answer:")
                answer_preview = response['answer'][:200]
                print(f"{answer_preview}{'...' if len(response['answer']) > 200 else ''}\n")
                print(f"⏱️ Time: {response['response_time']:.2f}s")
                print(f"📚 Sources: {len(response.get('sources', []))}")
                
                if response.get('sources'):
                    formats = {}
                    for doc in response['sources']:
                        fmt = doc.metadata.get('format', 'unknown')
                        formats[fmt] = formats.get(fmt, 0) + 1
                    print(f"📊 Formats: {formats}")
                
                print()
                
                result = {
                    'question': question['text'],
                    'category': question.get('category', 'uncategorized'),
                    'answer': response['answer'],
                    'response_time': response['response_time'],
                    'source_count': len(response.get('sources', [])),
                    'sources': [
                        {
                            'filename': doc.metadata.get('filename'),
                            'format': doc.metadata.get('format'),
                            'page': doc.metadata.get('page_number')
                        }
                        for doc in response.get('sources', [])[:5]
                    ]
                }
            
            self.results.append(result)
        
        total_elapsed = time.time() - total_start
        
        self._print_summary(total_elapsed)
        
        if not output_file:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            output_file = f"batch_results_{timestamp}.json"
        
        self._save_results(output_file)
    
    def _load_questions(self, filepath: str) -> List[Dict]:
        """Load questions from file."""
        path = Path(filepath)
        
        if not path.exists():
            print(f"❌ File not found: {filepath}")
            return []
        
        questions = []
        
        if filepath.endswith('.json'):
            try:
                with open(filepath, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                
                if isinstance(data, dict):
                    for category, qs in data.items():
                        for q in qs:
                            questions.append({'text': q, 'category': category})
                elif isinstance(data, list):
                    for q in data:
                        if isinstance(q, str):
                            questions.append({'text': q, 'category': 'uncategorized'})
                        elif isinstance(q, dict):
                            questions.append(q)
                return questions
            except json.JSONDecodeError:
                print("⚠️ Invalid JSON, trying as text...")
        
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                for line in f:
                    line = line.strip()
                    if line and not line.startswith('#'):
                        questions.append({'text': line, 'category': 'uncategorized'})
            return questions
        except Exception as e:
            print(f"❌ Error reading file: {e}")
            return []
    
    def _print_summary(self, total_time: float):
        """Print batch execution summary."""
        separator = "=" * 70
        
        print(separator)
        print("📊 BATCH EXECUTION SUMMARY")
        print(separator)
        
        if not self.results:
            return
        
        total = len(self.results)
        successful = sum(1 for r in self.results if 'answer' in r)
        failed = total - successful
        
        print(f"\nTotal questions: {total}")
        print(f"Successful: {successful}")
        print(f"Failed: {failed}")
        print(f"Total time: {total_time:.2f}s")
        
        if successful > 0:
            response_times = [r['response_time'] for r in self.results if 'answer' in r]
            avg_time = sum(response_times) / len(response_times)
            
            print("\nPerformance:")
            print(f"  Average time: {avg_time:.2f}s")
            print(f"  Fastest: {min(response_times):.2f}s")
            print(f"  Slowest: {max(response_times):.2f}s")
            
            categories = {}
            for r in self.results:
                cat = r.get('category', 'uncategorized')
                if cat not in categories:
                    categories[cat] = {'count': 0, 'total_time': 0}
                categories[cat]['count'] += 1
                categories[cat]['total_time'] += r.get('response_time', 0)
            
            if len(categories) > 1:
                print("\nBy category:")
                for cat, stats in categories.items():
                    avg = stats['total_time'] / stats['count']
                    print(f"  {cat}: {stats['count']} questions, avg {avg:.2f}s")
            
            total_sources = sum(r.get('source_count', 0) for r in self.results if 'answer' in r)
            print("\nRetrieval:")
            print(f"  Total sources: {total_sources}")
            print(f"  Avg per question: {total_sources / successful:.1f}")
        
        print(separator)
    
    def _save_results(self, output_file: str):
        """Save results to file."""
        try:
            output_data = {
                'metadata': {
                    'timestamp': datetime.now().isoformat(),
                    'total_questions': len(self.results),
                    'configuration': {
                        'chunk_size': self.chatbot.config.chunk_size,
                        'chunk_overlap': self.chatbot.config.chunk_overlap,
                        'embedding_model': self.chatbot.config.embedding_model,
                        'llm_model': os.path.basename(self.chatbot.config.model_path),
                    }
                },
                'results': self.results,
                'summary': self._generate_summary_dict()
            }
            
            with open(output_file, 'w', encoding='utf-8') as f:
                json.dump(output_data, f, indent=2, ensure_ascii=False)
            
            print(f"\n✅ Results saved to: {output_file}")
            
            text_file = Path(output_file).with_suffix('.txt')
            self._save_text_report(text_file)
            print(f"✅ Text report saved to: {text_file}")
            
        except Exception as e:
            print(f"❌ Error saving results: {e}")
    
    def _save_text_report(self, output_file: Path):
        """Save human-readable text report."""
        separator = "=" * 80
        dash_separator = "-" * 80
        
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(separator + "\n")
            f.write("BATCH QUERY RESULTS\n")
            f.write(separator + "\n\n")
            
            f.write(f"Timestamp: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
            f.write("Configuration:\n")
            f.write(f"  Chunk size: {self.chatbot.config.chunk_size}\n")
            f.write(f"  Chunk overlap: {self.chatbot.config.chunk_overlap}\n\n")
            f.write(separator + "\n\n")
            
            for i, result in enumerate(self.results, 1):
                f.write(dash_separator + "\n")
                f.write(f"[Q{i}] {result['question']}\n\n")
                
                if 'error' in result:
                    f.write(f"ERROR: {result['error']}\n")
                else:
                    f.write(f"Answer:\n{result['answer']}\n\n")
                    f.write(f"Time: {result['response_time']:.2f}s\n")
                    f.write(f"Sources: {result['source_count']}\n")
                
                f.write("\n")
    
    def _generate_summary_dict(self) -> Dict:
        """Generate summary statistics."""
        if not self.results:
            return {}
        
        successful = [r for r in self.results if 'answer' in r]
        
        summary = {
            'total_questions': len(self.results),
            'successful': len(successful),
            'failed': len(self.results) - len(successful),
        }
        
        if successful:
            response_times = [r['response_time'] for r in successful]
            summary['avg_response_time'] = sum(response_times) / len(response_times)
            summary['min_response_time'] = min(response_times)
            summary['max_response_time'] = max(response_times)
            
            total_sources = sum(r.get('source_count', 0) for r in successful)
            summary['avg_sources_per_question'] = total_sources / len(successful)
        
        return summary

    def _safe_delete_directory(self, path: str, max_attempts: int = 5) -> bool:
        """Safely delete directory with retry logic."""
        if not os.path.exists(path):
            return True
        
        for attempt in range(max_attempts):
            try:
                gc.collect()
                time.sleep(1 + attempt * 0.5)
                
                shutil.rmtree(path)
                return True
                
            except PermissionError:
                if attempt < max_attempts - 1:
                    if platform.system() == 'Windows':
                        try:
                            import subprocess
                            result = subprocess.run(
                                ['cmd', '/c', 'rmdir', '/s', '/q', path],
                                check=False,
                                capture_output=True,
                                timeout=5
                            )
                            if not os.path.exists(path):
                                return True
                        except:
                            pass
                else:
                    return False
            except Exception as e:
                logger.debug(f"Delete attempt {attempt + 1} failed: {e}")
                if attempt == max_attempts - 1:
                    return False
        
        return False
    
    def _cleanup_databases(self, database_paths: List[str]):
        """Clean up list of database directories."""
        if not database_paths:
            print("   ✓ No databases to clean")
            return
        
        print(f"   Found {len(database_paths)} database(s) to clean")
        print("   ⏳ Waiting for file handles to release...")
        gc.collect()
        time.sleep(3)
        
        cleaned = 0
        failed = []
        
        for db_path in database_paths:
            db_name = Path(db_path).name
            print(f"   Removing: {db_name}...", end=' ')
            
            if self._safe_delete_directory(db_path):
                print("✓")
                cleaned += 1
            else:
                print("✗")
                failed.append(db_path)
        
        if failed:
            print(f"\n   🔄 Retrying {len(failed)} failed deletion(s)...")
            time.sleep(5)
            gc.collect()
            
            for db_path in failed[:]:
                db_name = Path(db_path).name
                print(f"   Retry: {db_name}...", end=' ')
                
                if self._safe_delete_directory(db_path):
                    print("✓")
                    cleaned += 1
                    failed.remove(db_path)
                else:
                    print("✗")
        
        remaining = len(database_paths) - cleaned
        
        if remaining > 0:
            print(f"\n   ⚠️ {remaining} database(s) could not be deleted:")
            for db_path in failed:
                print(f"      • {Path(db_path).name}")
            print("   ℹ️ Run 'cleanup' command after closing chatbot to remove them")
        else:
            print(f"\n   ✅ All {cleaned} test database(s) cleaned successfully")

        self.embedding_model: str = "sentence-transformers/all-MiniLM-L6-v2"
        self.persist_directory: str = "./chroma_db"
        self.config_file: str = "chatbot_config.json"
    
    def get_formats_hash(self) -> str:
        if not self.selected_formats:
            return "default"
        format_values = sorted([f.value for f in self.selected_formats])
        format_string = "_".join(format_values)
        return hashlib.md5(format_string.encode()).hexdigest()[:8]
    
    def update_persist_directory(self):
        formats_hash = self.get_formats_hash()
        format_names = "_".join(sorted([f.value for f in self.selected_formats]))
        self.persist_directory = f"./chroma_db_{format_names}_{formats_hash}"
    
    def load_from_file(self, filepath: str = None) -> bool:
        filepath = filepath or self.config_file
        if not os.path.exists(filepath):
            return False
        try:
            with open(filepath, 'r') as f:
                config = json.load(f)
            self.model_path = config.get('model_path', '')
            self.documents_folder = config.get('documents_folder', '')
            format_strings = config.get('selected_formats', [])
            self.selected_formats = set()
            for fmt_str in format_strings:
                try:
                    fmt = DocumentFormat(fmt_str.lower())
                    self.selected_formats.add(fmt)
                except ValueError:
                    pass
            self.chunk_size = config.get('chunk_size', 1000)
            self.chunk_overlap = config.get('chunk_overlap', 200)
            self.embedding_model = config.get('embedding_model', self.embedding_model)
            self.update_persist_directory()
            return True
        except Exception as e:
            return False
    
    def save_to_file(self, filepath: str = None):
        filepath = filepath or self.config_file
        config = {
            'model_path': self.model_path,
            'documents_folder': self.documents_folder,
            'selected_formats': [fmt.value for fmt in self.selected_formats],
            'chunk_size': self.chunk_size,
            'chunk_overlap': self.chunk_overlap,
            'embedding_model': self.embedding_model,
            'persist_directory': self.persist_directory,
            'saved_at': datetime.now().isoformat()
        }
        try:
            with open(filepath, 'w') as f:
                json.dump(config, indent=2, fp=f)
        except Exception as e:
            logger.error(f"Failed to save: {e}")

    def _select_formats(self) -> Set[DocumentFormat]:
        available = [fmt for fmt in DocumentFormat if fmt.is_available()]
        if not available:
            print("❌ No document formats available!")
            return set()
        
        print("\nAvailable formats:")
        for i, fmt in enumerate(available, 1):
            print(f"  {i}. {fmt} - {fmt.get_description()}")
        
        while True:
            response = input("\nSelect formats (e.g., 1,2 or 'all'): ").strip().lower()
            if response == 'all':
                return set(available)
            try:
                indices = [int(x.strip()) for x in response.split(',')]
                selected = set()
                for idx in indices:
                    if 1 <= idx <= len(available):
                        selected.add(available[idx - 1])
                if selected:
                    return selected
                print("❌ No valid formats selected. Try again.")
            except ValueError:
                print("❌ Invalid input. Use format like: 1,2 or 'all'")
    
    def _get_folder_path(self, prompt: str, default: str = "") -> str:
        while True:
            if default:
                user_input = input(f"{prompt} [{default}]: ").strip()
                path = user_input if user_input else default
            else:
                path = input(f"{prompt}: ").strip()
            
            path = os.path.expanduser(path)
            if os.path.exists(path) and os.path.isdir(path):
                return os.path.abspath(path)
            elif self._get_yes_no("Folder doesn't exist. Create it?", default='y'):
                try:
                    os.makedirs(path, exist_ok=True)
                    return os.path.abspath(path)
                except Exception as e:
                    print(f"❌ Failed to create folder: {e}")
            else:
                print("❌ Please enter a valid folder path.")
    
    def _get_file_path(self, prompt: str, default: str = "", must_exist: bool = True) -> str:
        while True:
            if default:
                user_input = input(f"{prompt} [{default}]: ").strip()
                path = user_input if user_input else default
            else:
                path = input(f"{prompt}: ").strip()
            
            path = os.path.expanduser(path)
            if must_exist:
                if os.path.exists(path) and os.path.isfile(path):
                    return os.path.abspath(path)
                else:
                    print(f"❌ File not found: {path}")
                    print("   Please check the path and try again.")
            else:
                return os.path.abspath(path)
    
    def _get_yes_no(self, prompt: str, default: str = 'y') -> bool:
        choices = "Y/n" if default == 'y' else "y/N"
        while True:
            response = input(f"{prompt} [{choices}]: ").strip().lower()
            if not response:
                return default == 'y'
            if response in ('y', 'yes'):
                return True
            elif response in ('n', 'no'):
                return False
            print("❌ Please enter 'y' or 'n'")
    
    def _configure_advanced_settings(self):
        print("\n📏 Chunk Size (how large each text chunk is):")
        print("   Smaller = faster, less context")
        print("   Larger = slower, more context")
        self.chunk_size = self._get_integer("Chunk size", self.chunk_size, 100, 5000)
        
        print("\n🔗 Chunk Overlap (text shared between chunks):")
        print("   Higher = better context continuity, more redundancy")
        print("   Lower = less redundancy, faster")
        max_overlap = self.chunk_size // 2
        self.chunk_overlap = self._get_integer("Overlap", self.chunk_overlap, 0, max_overlap)
    
    def _get_integer(self, prompt: str, default: int, min_val: int = None, max_val: int = None) -> int:
        while True:
            user_input = input(f"{prompt} [{default}]: ").strip()
            if not user_input:
                return default
            try:
                value = int(user_input)
                if min_val is not None and value < min_val:
                    print(f"❌ Minimum value is {min_val}")
                    continue
                if max_val is not None and value > max_val:
                    print(f"❌ Maximum value is {max_val}")
                    continue
                return value
            except ValueError:
                print("❌ Please enter a valid integer.")
    
    def _validate_configuration(self) -> bool:
        issues = []
        if not self.model_path or not os.path.exists(self.model_path):
            issues.append("Model file not found")
        if not self.documents_folder:
            issues.append("Documents folder not set")
        if not self.selected_formats:
            issues.append("No formats selected")
        
        if issues:
            print("\n❌ Configuration Issues:")
            for issue in issues:
                print(f"  • {issue}")
            return False
        
        return True

class DocumentLoader:
    """Base document loader."""
    
    def __init__(self, file_path: str):
        self.file_path = Path(file_path)
        if not self.file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
    
    def load(self) -> List[Document]:
        raise NotImplementedError()
    
    def _create_metadata(self, additional: Dict = None) -> Dict:
        metadata = {
            'source': str(self.file_path),
            'filename': self.file_path.name,
            'file_type': self.file_path.suffix.lstrip('.'),
            'loaded_at': datetime.now().isoformat(),
        }
        if additional:
            metadata.update(additional)
        return metadata


class PlainTextLoader(DocumentLoader):
    """Plain text file loader."""
    
    def load(self) -> List[Document]:
        try:
            with open(self.file_path, 'r', encoding='utf-8') as f:
                content = f.read()
        except UnicodeDecodeError:
            with open(self.file_path, 'r', encoding='latin-1') as f:
                content = f.read()
        
        # Clean up excessive newlines
        content = re.sub(r'\n\s*\n\s*\n', '\n\n', content).strip()
        
        return [Document(
            page_content=content,
            metadata=self._create_metadata({'format': 'txt'})
        )]


class PDFLoader(DocumentLoader):
    """PDF document loader."""
    
    def load(self) -> List[Document]:
        documents = []
        try:
            import pdfplumber
            with pdfplumber.open(self.file_path) as pdf:
                for page_num, page in enumerate(pdf.pages, start=1):
                    text = page.extract_text()
                    if text and text.strip():
                        documents.append(Document(
                            page_content=text.strip(),
                            metadata=self._create_metadata({
                                'format': 'pdf',
                                'page_number': page_num,
                                'total_pages': len(pdf.pages)
                            })
                        ))
            return documents
        except Exception as e:
            raise Exception(f"Failed to load PDF: {e}")


class HTMLLoader(DocumentLoader):
    """HTML document loader with Markdown conversion."""
    
    def __init__(self, file_path: str):
        super().__init__(file_path)
        self.h = html2text.HTML2Text()
        self.h.ignore_links = False
        self.h.body_width = 0
    
    def load(self) -> List[Document]:
        try:
            with open(self.file_path, 'r', encoding='utf-8') as f:
                html_content = f.read()
        except UnicodeDecodeError:
            with open(self.file_path, 'r', encoding='latin-1') as f:
                html_content = f.read()
        
        soup = BeautifulSoup(html_content, 'html.parser')
        
        # Remove unwanted tags
        for tag in soup.find_all(['script', 'style', 'nav', 'footer', 'header']):
            tag.decompose()
        
        # Convert to Markdown
        markdown = self.h.handle(str(soup)).strip()
        
        return [Document(
            page_content=markdown,
            metadata=self._create_metadata({'format': 'html'})
        )]


class EnhancedDITALoader(DocumentLoader):
    """Enhanced DITA loader that converts to Markdown."""
    
    def __init__(self, file_path: str):
        super().__init__(file_path)
        self.note_icons = {
            'note': '📝',
            'tip': '💡',
            'important': '❗',
            'warning': '⚠️',
            'caution': '🚨',
            'danger': '☠️',
            'attention': '⚡',
            'remember': '🔔',
            'restriction': '🚫',
            'fastpath': '⚡',
            'other': 'ℹ️'
        }
    
    def load(self) -> List[Document]:
        """Load and convert DITA to Markdown."""
        try:
            tree = ET.parse(self.file_path)
            root = tree.getroot()
            
            topic_type = self._get_topic_type(root)
            markdown_content = self._convert_to_markdown(root, topic_type)
            metadata = self._extract_metadata(root, topic_type)
            
            return [Document(
                page_content=markdown_content.strip(),
                metadata=self._create_metadata(metadata)
            )]
            
        except ET.ParseError as e:
            raise Exception(f"Failed to parse DITA XML: {e}")
        except Exception as e:
            raise Exception(f"Failed to load DITA: {e}")
    
    def _get_topic_type(self, root: ET.Element) -> str:
        """Determine DITA topic type."""
        tag = root.tag.lower()
        if '}' in tag:
            tag = tag.split('}')[1]
        
        topic_types = {
            'concept': 'Concept',
            'task': 'Task',
            'reference': 'Reference',
            'topic': 'Topic',
            'troubleshooting': 'Troubleshooting',
            'glossentry': 'Glossary'
        }
        
        return topic_types.get(tag, 'Topic')
    
    def _convert_to_markdown(self, element: ET.Element, topic_type: str, level: int = 0) -> str:
        """Convert DITA XML to Markdown recursively."""
        markdown = []
        tag = self._get_tag_name(element)
        
        if tag in ('concept', 'task', 'reference', 'topic', 'troubleshooting'):
            markdown.append(self._process_topic(element, topic_type))
        elif tag == 'title':
            markdown.append(self._process_title(element, level))
        elif tag in ('p', 'shortdesc'):
            markdown.append(self._process_paragraph(element))
        elif tag == 'ul':
            markdown.append(self._process_unordered_list(element))
        elif tag == 'ol':
            markdown.append(self._process_ordered_list(element))
        elif tag == 'note':
            markdown.append(self._process_note(element))
        elif tag == 'codeblock':
            markdown.append(self._process_codeblock(element))
        elif tag == 'table':
            markdown.append(self._process_table(element))
        elif tag == 'section':
            markdown.append(self._process_section(element, level + 1))
        elif tag == 'xref':
            markdown.append(self._process_xref(element))
        elif tag in ('b', 'strong'):
            markdown.append(f"**{self._get_text_content(element)}**")
        elif tag in ('i', 'em'):
            markdown.append(f"*{self._get_text_content(element)}*")
        elif tag == 'codeph':
            markdown.append(f"`{self._get_text_content(element)}`")
        elif tag == 'filepath':
            markdown.append(f"`{self._get_text_content(element)}`")
        elif tag == 'cmdname':
            markdown.append(f"**`{self._get_text_content(element)}`**")
        elif tag == 'parmname':
            markdown.append(f"`{self._get_text_content(element)}`")
        elif tag == 'apiname':
            markdown.append(f"`{self._get_text_content(element)}`")
        elif tag == 'userinput':
            code_content = self._get_text_content(element)
            markdown.append(f"\n{code_content}\n")
        elif tag == 'systemoutput':
            code_content = self._get_text_content(element)
            markdown.append(f"\n{code_content}\n")
        elif tag in ('taskbody', 'conbody', 'refbody', 'body'):
            markdown.append(self._process_body(element, level + 1))
        elif tag == 'prereq':
            markdown.append(self._process_prereq(element))
        elif tag == 'steps':
            markdown.append(self._process_steps(element))
        elif tag == 'steps-unordered':
            markdown.append(self._process_steps_unordered(element))
        elif tag == 'result':
            markdown.append(self._process_result(element))
        elif tag == 'example':
            markdown.append(self._process_example(element))
        elif tag == 'fig':
            markdown.append(self._process_figure(element))
        elif tag == 'image':
            markdown.append(self._process_image(element))
        elif tag == 'dl':
            markdown.append(self._process_definition_list(element))
        elif tag == 'simpletable':
            markdown.append(self._process_simple_table(element))
        elif tag in ('prolog', 'metadata', 'related-links'):
            pass
        else:
            for child in element:
                markdown.append(self._convert_to_markdown(child, topic_type, level))
        
        return '\n'.join(filter(None, markdown))
    
    def _process_topic(self, element: ET.Element, topic_type: str) -> str:
        """Process main topic element."""
        markdown = []
        markdown.append(f"**[{topic_type}]**\n")
        
        title = element.find('.//title')
        if title is not None:
            markdown.append(f"# {self._get_text_content(title)}\n")
        
        shortdesc = element.find('.//shortdesc')
        if shortdesc is not None:
            markdown.append(f"*{self._get_text_content(shortdesc)}*\n")
        
        for child in element:
            tag = self._get_tag_name(child)
            if tag not in ('title', 'shortdesc', 'prolog'):
                markdown.append(self._convert_to_markdown(child, topic_type, 1))
        
        return '\n'.join(filter(None, markdown))
    
    def _process_title(self, element: ET.Element, level: int) -> str:
        heading = '#' * min(level + 1, 6)
        return f"{heading} {self._get_text_content(element)}\n"
    
    def _process_paragraph(self, element: ET.Element) -> str:
        text = self._get_mixed_content(element)
        return f"{text}\n" if text.strip() else ""
    
    def _process_unordered_list(self, element: ET.Element) -> str:
        items = []
        for li in element.findall('.//li'):
            content = self._get_mixed_content(li)
            if content.strip():
                items.append(f"- {content}")
        return '\n'.join(items) + '\n' if items else ""
    
    def _process_ordered_list(self, element: ET.Element) -> str:
        items = []
        for i, li in enumerate(element.findall('.//li'), 1):
            content = self._get_mixed_content(li)
            if content.strip():
                items.append(f"{i}. {content}")
        return '\n'.join(items) + '\n' if items else ""
    
    def _process_note(self, element: ET.Element) -> str:
        note_type = element.get('type', 'note').lower()
        icon = self.note_icons.get(note_type, self.note_icons['other'])
        type_label = note_type.upper()
        content = self._get_mixed_content(element)
        return f"{icon} **{type_label}:** {content}\n"
    
    def _process_codeblock(self, element: ET.Element) -> str:
        language = element.get('outputclass', '')
        code = element.text or ""
        code = code.strip()
        
        if language:
            return f"{language}\n{code}\n"
        else:
            return f"\n{code}\n"
        
    def _process_table(self, element: ET.Element) -> str:
        markdown = []
        tgroup = element.find('.//tgroup')
        if tgroup is None:
            return ""
        
        thead = tgroup.find('.//thead')
        headers = []
        if thead is not None:
            row = thead.find('.//row')
            if row is not None:
                for entry in row.findall('.//entry'):
                    headers.append(self._get_text_content(entry))
        
        if not headers:
            cols = int(tgroup.get('cols', '2'))
            headers = [f"Column {i+1}" for i in range(cols)]
        
        markdown.append('| ' + ' | '.join(headers) + ' |')
        markdown.append('| ' + ' | '.join(['---'] * len(headers)) + ' |')
        
        tbody = tgroup.find('.//tbody')
        if tbody is not None:
            for row in tbody.findall('.//row'):
                cells = []
                for entry in row.findall('.//entry'):
                    cells.append(self._get_text_content(entry))
                while len(cells) < len(headers):
                    cells.append('')
                markdown.append('| ' + ' | '.join(cells) + ' |')
        
        return '\n'.join(markdown) + '\n'
    
    def _process_simple_table(self, element: ET.Element) -> str:
        markdown = []
        sthead = element.find('.//sthead')
        headers = []
        if sthead is not None:
            for stentry in sthead.findall('.//stentry'):
                headers.append(self._get_text_content(stentry))
        
        if headers:
            markdown.append('| ' + ' | '.join(headers) + ' |')
            markdown.append('| ' + ' | '.join(['---'] * len(headers)) + ' |')
        
        for strow in element.findall('.//strow'):
            cells = []
            for stentry in strow.findall('.//stentry'):
                cells.append(self._get_text_content(stentry))
            if cells:
                markdown.append('| ' + ' | '.join(cells) + ' |')
        
        return '\n'.join(markdown) + '\n' if markdown else ""
    
    def _process_section(self, element: ET.Element, level: int) -> str:
        markdown = []
        title = element.find('.//title')
        if title is not None:
            heading = '#' * min(level + 1, 6)
            markdown.append(f"{heading} {self._get_text_content(title)}\n")
        
        for child in element:
            if self._get_tag_name(child) != 'title':
                markdown.append(self._convert_to_markdown(child, '', level))
        
        return '\n'.join(filter(None, markdown))
    
    def _process_body(self, element: ET.Element, level: int) -> str:
        markdown = []
        for child in element:
            markdown.append(self._convert_to_markdown(child, '', level))
        return '\n'.join(filter(None, markdown))
    
    def _process_prereq(self, element: ET.Element) -> str:
        content = self._get_mixed_content(element)
        return f"### Prerequisites\n\n{content}\n"
    
    def _process_steps(self, element: ET.Element) -> str:
        markdown = ["### Steps\n"]
        
        for i, step in enumerate(element.findall('.//step'), 1):
            cmd = step.find('.//cmd')
            if cmd is not None:
                markdown.append(f"{i}. **{self._get_text_content(cmd)}**")
                
                for info in step.findall('.//info'):
                    info_text = self._get_mixed_content(info)
                    if info_text.strip():
                        markdown.append(f"   {info_text}")
                
                substeps = step.find('.//substeps')
                if substeps is not None:
                    for j, substep in enumerate(substeps.findall('.//substep'), 1):
                        subcmd = substep.find('.//cmd')
                        if subcmd is not None:
                            markdown.append(f"   {chr(97+j-1)}. {self._get_text_content(subcmd)}")
                
                stepresult = step.find('.//stepresult')
                if stepresult is not None:
                    markdown.append(f"   *Result:* {self._get_text_content(stepresult)}")
                
                markdown.append("")
        
        return '\n'.join(markdown)
    
    def _process_steps_unordered(self, element: ET.Element) -> str:
        markdown = ["### Steps (any order)\n"]
        
        for step in element.findall('.//step'):
            cmd = step.find('.//cmd')
            if cmd is not None:
                markdown.append(f"- **{self._get_text_content(cmd)}**")
                
                for info in step.findall('.//info'):
                    info_text = self._get_mixed_content(info)
                    if info_text.strip():
                        markdown.append(f"  {info_text}")
                
                markdown.append("")
        
        return '\n'.join(markdown)
    
    def _process_result(self, element: ET.Element) -> str:
        content = self._get_mixed_content(element)
        return f"### Result\n\n{content}\n"
    
    def _process_example(self, element: ET.Element) -> str:
        markdown = ["### Example\n"]
        
        title = element.find('.//title')
        if title is not None:
            markdown.append(f"**{self._get_text_content(title)}**\n")
        
        for child in element:
            if self._get_tag_name(child) != 'title':
                markdown.append(self._convert_to_markdown(child, '', 3))
        
        return '\n'.join(filter(None, markdown))
    
    def _process_figure(self, element: ET.Element) -> str:
        markdown = []
        title = element.find('.//title')
        if title is not None:
            markdown.append(f"**Figure:** {self._get_text_content(title)}\n")
        
        image = element.find('.//image')
        if image is not None:
            markdown.append(self._process_image(image))
        
        return '\n'.join(filter(None, markdown))
    
    def _process_image(self, element: ET.Element) -> str:
        href = element.get('href', '')
        alt = element.get('alt', 'Image')
        return f"![{alt}]({href})\n"
    
    def _process_definition_list(self, element: ET.Element) -> str:
        markdown = []
        
        for dlentry in element.findall('.//dlentry'):
            dt = dlentry.find('.//dt')
            dd = dlentry.find('.//dd')
            
            if dt is not None:
                term = self._get_text_content(dt)
                definition = self._get_text_content(dd) if dd is not None else ""
                
                markdown.append(f"**{term}**")
                if definition:
                    markdown.append(f": {definition}\n")
        
        return '\n'.join(markdown)
    
    def _process_xref(self, element: ET.Element) -> str:
        href = element.get('href', '')
        text = self._get_text_content(element) or href
        
        if href:
            return f"[{text}]({href})"
        else:
            return text
    
    def _get_mixed_content(self, element: ET.Element) -> str:
        parts = []
        
        if element.text:
            parts.append(element.text.strip())
        
        for child in element:
            tag = self._get_tag_name(child)
            
            if tag in ('b', 'strong'):
                parts.append(f"**{self._get_text_content(child)}**")
            elif tag in ('i', 'em'):
                parts.append(f"*{self._get_text_content(child)}*")
            elif tag == 'codeph':
                parts.append(f"`{self._get_text_content(child)}`")
            elif tag == 'xref':
                parts.append(self._process_xref(child))
            elif tag == 'cmdname':
                parts.append(f"**`{self._get_text_content(child)}`**")
            elif tag == 'parmname':
                parts.append(f"`{self._get_text_content(child)}`")
            elif tag == 'filepath':
                parts.append(f"`{self._get_text_content(child)}`")
            else:
                parts.append(self._get_text_content(child))
            
            if child.tail:
                parts.append(child.tail.strip())
        
        return ' '.join(filter(None, parts))
    
    def _get_text_content(self, element: ET.Element) -> str:
        if element is None:
            return ""
        
        text_parts = []
        
        if element.text:
            text_parts.append(element.text)
        
        for child in element:
            text_parts.append(self._get_text_content(child))
            if child.tail:
                text_parts.append(child.tail)
        
        return ''.join(text_parts).strip()
    
    def _get_tag_name(self, element: ET.Element) -> str:
        tag = element.tag
        if '}' in tag:
            tag = tag.split('}')[1]
        return tag.lower()
    
    def _extract_metadata(self, root: ET.Element, topic_type: str) -> Dict:
        metadata = {
            'format': 'dita',
            'topic_type': topic_type,
            'topic_id': root.get('id', ''),
        }
        
        prolog = root.find('.//prolog')
        if prolog is not None:
            author = prolog.find('.//author')
            if author is not None:
                metadata['author'] = self._get_text_content(author)
            
            keywords = []
            for keyword in prolog.findall('.//keyword'):
                kw = self._get_text_content(keyword)
                if kw:
                    keywords.append(kw)
            if keywords:
                metadata['keywords'] = ', '.join(keywords)
            
            audience = prolog.find('.//audience')
            if audience is not None:
                aud_type = audience.get('type', '')
                aud_job = audience.get('job', '')
                if aud_type or aud_job:
                    metadata['audience'] = f"{aud_type} {aud_job}".strip()
            
            created = prolog.find('.//created')
            if created is not None:
                metadata['created_date'] = created.get('date', '')
            
            revised = prolog.find('.//revised')
            if revised is not None:
                metadata['revised_date'] = revised.get('modified', '')
        
        return metadata


class DocumentLoaderFactory:
    """Factory for creating document loaders."""
    
    _loaders = {
        DocumentFormat.PLAIN_TEXT: PlainTextLoader,
        DocumentFormat.PDF: PDFLoader,
        DocumentFormat.HTML: HTMLLoader,
        DocumentFormat.DITA: EnhancedDITALoader,
    }
    
    @classmethod
    def create_loader(cls, file_path: str) -> DocumentLoader:
        path = Path(file_path)
        doc_format = DocumentFormat.from_extension(path.suffix)
        
        if doc_format is None:
            raise ValueError(f"Unsupported file type: {path.suffix}")
        
        if not doc_format.is_available():
            raise ValueError(f"Support for {doc_format} is not installed")
        
        loader_class = cls._loaders.get(doc_format)
        if loader_class is None:
            raise ValueError(f"No loader available for {doc_format}")
        
        return loader_class(file_path)


class MultiFormatTextSplitter:
    """Text splitter optimized for different document formats."""
    
    def __init__(self, chunk_size: int = 1000, chunk_overlap: int = 200):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
    
    def split_documents(self, documents: List[Document]) -> List[Document]:
        all_chunks = []
        
        for doc in documents:
            doc_format = doc.metadata.get('format', 'unknown')
            separators = self._get_separators(doc_format)
            
            splitter = RecursiveCharacterTextSplitter(
                chunk_size=self.chunk_size,
                chunk_overlap=self.chunk_overlap,
                separators=separators,
                length_function=len,
            )
            
            chunks = splitter.split_documents([doc])
            
            for i, chunk in enumerate(chunks):
                chunk.metadata['chunk_id'] = len(all_chunks) + i
                chunk.metadata['chunk_index'] = i
                chunk.metadata['total_chunks'] = len(chunks)
            
            all_chunks.extend(chunks)
        
        return all_chunks

    def verify_chunk_overlap(self, documents: List[Document]) -> None:
        """Diagnostic function to verify chunk overlap."""
        print("\n🔍 CHUNK OVERLAP VERIFICATION")
        print("=" * 70)
        
        doc_chunks = {}
        for doc in documents:
            filename = doc.metadata.get('filename', 'unknown')
            if filename not in doc_chunks:
                doc_chunks[filename] = []
            doc_chunks[filename].append(doc)
        
        for filename, chunks in doc_chunks.items():
            if len(chunks) > 1:
                print(f"\n📄 {filename}: {len(chunks)} chunks")
                
                for i in range(len(chunks) - 1):
                    chunk1_text = chunks[i].page_content
                    chunk2_text = chunks[i + 1].page_content
                    
                    # Check for overlap
                    overlap_found = False
                    overlap_length = 0
                    
                    # Check last N characters of chunk1 against first N characters of chunk2
                    for length in range(min(len(chunk1_text), len(chunk2_text), self.chunk_overlap), 10, -1):
                        if chunk1_text[-length:] == chunk2_text[:length]:
                            overlap_found = True
                            overlap_length = length
                            break
                    
                    if overlap_found:
                        print(f"   ✅ Chunks {i+1}↔{i+2}: {overlap_length} chars overlap")
                    else:
                        print(f"   ❌ Chunks {i+1}↔{i+2}: NO overlap detected")
                        print(f"      Chunk {i+1} ends: ...{chunk1_text[-50:]}")
                        print(f"      Chunk {i+2} starts: {chunk2_text[:50]}...")
            else:
                print(f"\n📄 {filename}: Single chunk (no overlap needed)")
        
        print("\n" + "=" * 70)
    
    def _get_separators(self, doc_format: str) -> List[str]:
        if doc_format in ('html', 'dita'):
            return [
                "\n## ",
                "\n### ",
                "\n#### ",
                "\n\n",
                "\n",
                ". ",
                " ",
                ""
            ]
        else:
            return [
                "\n\n",
                "\n",
                ". ",
                " ",
                ""
            ]

class InteractiveChatbot:
    """Main chatbot class with RAG capabilities."""
    
    def __init__(self, config: UserConfiguration):
        self.config = config
        self.config.update_persist_directory()
        
        self.vectorstore = None
        self.qa_chain = None
        self.embeddings = None
        self.llm = None
        self.retriever = None
        
        self.loaded_documents = []
        self.document_stats = {}
        
        self.session_logger = SessionLogger(config)
        
        self.query_cache = {}
        self.query_log = []
        self.llm_config = {}
        self.retriever_config = {}
    
    def load_documents(self) -> List[Document]:
        """Load documents from configured folder."""
        logger.info("📁 LOADING DOCUMENTS")
        
        documents_path = Path(self.config.documents_folder)
        selected_extensions = set()
        for fmt in self.config.selected_formats:
            selected_extensions.update(fmt.get_extensions())
        
        all_files = []
        for ext in selected_extensions:
            all_files.extend(documents_path.rglob(f"*{ext}"))
        
        logger.info(f"Found {len(all_files)} file(s) to process")
        
        documents = []
        stats = {str(fmt): {'count': 0, 'success': 0, 'failed': 0} 
                for fmt in self.config.selected_formats}
        
        for file_path in all_files:
            try:
                doc_format = DocumentFormat.from_extension(file_path.suffix)
                if doc_format not in self.config.selected_formats:
                    continue
                
                fmt_str = str(doc_format)
                stats[fmt_str]['count'] += 1
                
                logger.info(f"Loading: {file_path.name}")
                loader = DocumentLoaderFactory.create_loader(str(file_path))
                file_docs = loader.load()
                documents.extend(file_docs)
                stats[fmt_str]['success'] += 1
                
            except Exception as e:
                if doc_format is not None and doc_format in self.config.selected_formats:
                    stats[str(doc_format)]['failed'] += 1
                logger.error(f"Failed to load {file_path.name}: {e}")
        
        self.document_stats = stats
        
        logger.info(f"Loaded {len(documents)} document(s)")
        
        splitter = MultiFormatTextSplitter(
            chunk_size=self.config.chunk_size,
            chunk_overlap=self.config.chunk_overlap
        )
        chunks = splitter.split_documents(documents)

        splitter.verify_chunk_overlap(chunks)
        
        logger.info(f"✅ Created {len(chunks)} chunks")
        return chunks
    
    def create_vectorstore(self, documents: List[Document]):
        """Create or load vector database."""
        logger.info("🔍 VECTOR STORE SETUP")
        
        self.embeddings = HuggingFaceEmbeddings(
            model_name=self.config.embedding_model,
            model_kwargs={'device': 'cpu'},
            encode_kwargs={'normalize_embeddings': True}
        )
        
        db_exists = os.path.exists(self.config.persist_directory)
        
        if db_exists:
            try:
                logger.info("Checking existing database...")
                test_store = Chroma(
                    persist_directory=self.config.persist_directory,
                    embedding_function=self.embeddings
                )
                
                sample = test_store.get(limit=10)
                if sample and 'metadatas' in sample:
                    stored_formats = {m['format'] for m in sample['metadatas'] if m and 'format' in m}
                    selected_formats = {f.value for f in self.config.selected_formats}
                    
                    if stored_formats == selected_formats:
                        logger.info("✅ Reusing existing database")
                        self.vectorstore = test_store
                        return
                    else:
                        logger.info("Format mismatch, recreating database")
                        shutil.rmtree(self.config.persist_directory)
            except Exception as e:
                logger.warning(f"Database check failed: {e}")
                if os.path.exists(self.config.persist_directory):
                    shutil.rmtree(self.config.persist_directory)
        
        logger.info("Creating new vector database...")
        self.vectorstore = Chroma.from_documents(
            documents=documents,
            embedding=self.embeddings,
            persist_directory=self.config.persist_directory
        )
        logger.info("✅ Database created")
    
    def setup_llm(self):
        """Setup language model."""
        logger.info("🤖 LOADING LLM")
        
        model_name = os.path.basename(self.config.model_path).lower()
        if 'mistral' in model_name:
            model_type = 'mistral'
        elif 'llama' in model_name:
            model_type = 'llama'
        else:
            model_type = 'llama'
        
        self.llm_config = {
            'max_new_tokens': 512,
            'temperature': 0.1,
            'context_length': 2048,
            'top_p': 0.95,
            'repetition_penalty': 1.1,
        }
        
        self.llm = CTransformers(
            model=self.config.model_path,
            model_type=model_type,
            config=self.llm_config
        )
        logger.info("✅ LLM loaded")
    
    def create_qa_chain(self):
        """Create question-answering chain."""
        self.setup_llm()
        
        template = """You are a helpful AI assistant. Use the following context to answer the question accurately and concisely.

Context:
{context}

Question: {question}

Answer:"""
        
        prompt = PromptTemplate(
            template=template,
            input_variables=["context", "question"]
        )
        
        self.retriever_config = {
            'search_type': 'similarity',
            'k': 5
        }
        
        self.retriever = self.vectorstore.as_retriever(
            search_type=self.retriever_config['search_type'],
            search_kwargs={"k": self.retriever_config['k']}
        )
        
        def format_docs(docs):
            return "\n\n".join(doc.page_content for doc in docs)
        
        self.qa_chain = (
            {"context": self.retriever | format_docs, "question": RunnablePassthrough()}
            | prompt
            | self.llm
            | StrOutputParser()
        )
        
        self.session_logger.set_model_params(self.llm_config, self.retriever_config)
    
    def initialize(self):
        """Initialize chatbot."""
        logger.info("🚀 INITIALIZING CHATBOT")
        
        documents = self.load_documents()
        self.session_logger.set_document_stats(self.document_stats)
        
        self.create_vectorstore(documents)
        self.create_qa_chain()
        
        logger.info("✅ CHATBOT READY")
    
    def query(self, question: str) -> Dict:
        """Process a question and return answer."""
        if not self.qa_chain:
            raise Exception("Chatbot not initialized")
        
        start_time = time.time()
        
        try:
            source_docs = self.retriever.invoke(question)
            answer = self.qa_chain.invoke(question)
            elapsed = time.time() - start_time
            
            response = {
                "answer": answer,
                "sources": source_docs,
                "source_chunks": [doc.page_content for doc in source_docs],  # ADD THIS LINE
                "response_time": elapsed,
                "from_cache": False
            }
            
            self.session_logger.log_query(
                question=question,
                answer=answer,
                response_time=elapsed,
                sources=source_docs,
                from_cache=False
            )
            
            return response
            
        except Exception as e:
            logger.error(f"Query failed: {e}")
            return {"error": str(e)}
    
    def finalize_session(self):
        """Finalize session and save logs."""
        self.session_logger.finalize()

                            print(f"✅ Saved to: {output_file}")
                        
                    except Exception as e:
                        print(f"❌ Error: {e}")
                        logger.exception("View document failed")
                    
                    continue
                
                # ============================================================
                # COMMAND: EXPORT SPECIFIC DOCUMENT
                # ============================================================
                if question.lower().startswith('export ') and not question.lower().startswith('export-'):
                    parts = question.split(maxsplit=1)
                    
                    if len(parts) < 2:
                        print("\n❌ Usage: export <filename>")
                        print("\nExamples:")
                        print("  export installation.html")
                        print("  export user_guide.dita")
                        continue
                    
                    filename = parts[1].strip()
                    
                    try:
                        # Query vectorstore
                        results = chatbot.vectorstore.get(
                            where={"filename": filename}
                        )
                        
                        if not results or not results['documents']:
                            print(f"\n❌ Document not found: {filename}")
                            print("\n💡 Use 'list-docs' to see available files")
                            continue
                        
                        # Create output directory
                        output_dir = Path("./markdown_exports")
                        output_dir.mkdir(exist_ok=True)
                        
                        # Create safe filename
                        safe_name = filename.replace('.html', '').replace('.dita', '').replace('.htm', '')
                        safe_name = re.sub(r'[^\w\-_]', '_', safe_name)
                        output_file = output_dir / f"{safe_name}.md"
                        
                        # Get metadata
                        metadata = results['metadatas'][0]
                        
                        # Write file
                        with open(output_file, 'w', encoding='utf-8') as f:
                            # Header
                            f.write(f"# {filename}\n\n")
                            f.write(f"**Format:** {metadata.get('format', 'unknown').upper()}\n")
                            f.write(f"**Chunks:** {len(results['documents'])}\n")
                            
                            if metadata.get('topic_type'):
                                f.write(f"**Type:** {metadata['topic_type']}\n")
                            
                            if metadata.get('total_pages'):
                                f.write(f"**Pages:** {metadata['total_pages']}\n")
                            
                            if metadata.get('author'):
                                f.write(f"**Author:** {metadata['author']}\n")
                            
                            if metadata.get('keywords'):
                                f.write(f"**Keywords:** {metadata['keywords']}\n")
                            
                            f.write(f"\n**Exported:** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                            f.write("\n---\n\n")
                            
                            # Content
                            if len(results['documents']) == 1:
                                f.write(results['documents'][0])
                            else:
                                for i, chunk in enumerate(results['documents'], 1):
                                    f.write(f"## Chunk {i} of {len(results['documents'])}\n\n")
                                    f.write(chunk)
                                    if i < len(results['documents']):
                                        f.write("\n\n---\n\n")
                        
                        print(f"\n✅ Exported successfully!")
                        print(f"   File: {output_file}")
                        print(f"   Size: {output_file.stat().st_size:,} bytes")
                        print(f"   Chunks: {len(results['documents'])}")
                        
                    except Exception as e:
                        print(f"❌ Error: {e}")
                        logger.exception("Export document failed")
                    
                    continue
                
                # ============================================================
                # COMMAND: EXPORT ALL DOCUMENTS
                # ============================================================
                if question.lower() == 'export-all':
                    print("\n📦 EXPORTING ALL DOCUMENTS")
                    print("=" * 70)
                    
                    try:
                        # Create output directory
                        output_dir = Path("./markdown_exports")
                        output_dir.mkdir(exist_ok=True)
                        
                        # Get all documents
                        all_docs = chatbot.vectorstore.get()
                        
                        if not all_docs or not all_docs['documents']:
                            print("❌ No documents to export")
                            continue
                        
                        # Group by filename
                        files = {}
                        for i, (doc, metadata) in enumerate(zip(all_docs['documents'], all_docs['metadatas'])):
                            filename = metadata.get('filename', f'unknown_{i}')
                            
                            if filename not in files:
                                files[filename] = {
                                    'chunks': [],
                                    'metadata': metadata
                                }
                            
                            files[filename]['chunks'].append(doc)
                        
                        print(f"Found {len(files)} document(s) to export\n")
                        
                        # Export each file
                        exported = 0
                        total_size = 0
                        
                        for filename, data in sorted(files.items()):
                            # Create safe filename
                            safe_name = filename.replace('.html', '').replace('.dita', '').replace('.htm', '')
                            safe_name = re.sub(r'[^\w\-_]', '_', safe_name)
                            output_file = output_dir / f"{safe_name}.md"
                            
                            # Write file
                            with open(output_file, 'w', encoding='utf-8') as f:
                                # Header
                                f.write(f"# {filename}\n\n")
                                f.write(f"**Format:** {data['metadata'].get('format', 'unknown').upper()}\n")
                                f.write(f"**Chunks:** {len(data['chunks'])}\n")
                                
                                if data['metadata'].get('topic_type'):
                                    f.write(f"**Type:** {data['metadata']['topic_type']}\n")
                                
                                if data['metadata'].get('author'):
                                    f.write(f"**Author:** {data['metadata']['author']}\n")
                                
                                f.write(f"\n**Exported:** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                                f.write("\n---\n\n")
                                
                                # Content
                                if len(data['chunks']) == 1:
                                    f.write(data['chunks'][0])
                                else:
                                    for i, chunk in enumerate(data['chunks'], 1):
                                        f.write(f"## Chunk {i} of {len(data['chunks'])}\n\n")
                                        f.write(chunk)
                                        if i < len(data['chunks']):
                                            f.write("\n\n---\n\n")
                            
                            file_size = output_file.stat().st_size
                            total_size += file_size
                            
                            exported += 1
                            print(f"  ✓ {filename}")
                            print(f"    → {output_file.name} ({file_size:,} bytes)")
                        
                        print(f"\n{'='*70}")
                        print(f"✅ Export complete!")
                        print(f"   Files:      {exported}")
                        print(f"   Location:   {output_dir.absolute()}")
                        print(f"   Total size: {total_size:,} bytes")
                        print("=" * 70)
                        
                    except Exception as e:
                        print(f"❌ Error: {e}")
                        logger.exception("Export all failed")
                    
                    continue
                
                # ============================================================
                # COMMAND: CLEANUP
                # ============================================================
                if question.lower() == 'cleanup':
                    comparison = ConfigurationComparison(chatbot.config)
                    comparison.cleanup_old_databases()
                    continue
                
                # ============================================================
                # COMMAND: BATCH MODE
                # ============================================================
                if question.lower().startswith('batch'):
                    parts = question.split(maxsplit=1)
                    if len(parts) < 2:
                        print("\n❌ Usage: batch <questions_file>")
                        print("\nExample: batch questions.txt")
                        continue
                    
                    questions_file = parts[1].strip()
                    output = input("Output file (press Enter for auto): ").strip()
                    
                    runner = BatchQueryRunner(chatbot)
                    runner.run_from_file(questions_file, output if output else None)
                    continue
                
                # ============================================================
                # COMMAND: COMPARE CONFIGURATIONS
                # ============================================================
                if question.lower().startswith('compare-configs'):
                    parts = question.split(maxsplit=1)
                    if len(parts) < 2:
                        print("\n❌ Usage: compare-configs <questions_file>")
                        continue
                    
                    questions_file = parts[1].strip()
                    
                    # Ask about cleanup
                    print("\n🧹 Clean up old test databases first?")
                    if input("Clean old databases? [Y/n]: ").strip().lower() in ('', 'y', 'yes'):
                        comparison = ConfigurationComparison(chatbot.config)
                        comparison.cleanup_old_databases()
                    
                    print("\nTest modes:")
                    print("  1. Quick (3 configs: small/medium/large)")
                    print("  2. Comprehensive (8 configs: sizes + overlaps)")
                    print("  3. Size focused (7 sizes, fixed 20% overlap)")
                    print("  4. Overlap focused (7 overlaps, fixed 1000 size)")
                    print("  5. Matrix (12 configs: 3 sizes × 4 overlaps)")
                    
                    choice = input("\nChoice [1]: ").strip() or "1"
                    
                    mode_map = {
                        "1": "quick",
                        "2": "comprehensive",
                        "3": "size_focused",
                        "4": "overlap_focused",
                        "5": "matrix"
                    }
                    
                    test_mode = mode_map.get(choice, "quick")
                    
                    comparison = ConfigurationComparison(chatbot.config)
                    comparison.run_comparison(questions_file, test_mode=test_mode)
                    continue
                
                # ============================================================
                # COMMAND: TEST SIZES
                # ============================================================
                if question.lower().startswith('test-sizes'):
                    parts = question.split(maxsplit=1)
                    if len(parts) < 2:
                        print("\n❌ Usage: test-sizes <questions_file>")
                        continue
                    
                    comparison = ConfigurationComparison(chatbot.config)
                    comparison.run_comparison(parts[1].strip(), test_mode="size_focused")
                    continue
                
                # ============================================================
                # COMMAND: TEST OVERLAPS
                # ============================================================
                if question.lower().startswith('test-overlaps'):
                    parts = question.split(maxsplit=1)
                    if len(parts) < 2:
                        print("\n❌ Usage: test-overlaps <questions_file>")
                        continue
                    
                    comparison = ConfigurationComparison(chatbot.config)
                    comparison.run_comparison(parts[1].strip(), test_mode="overlap_focused")
                    continue
                
                # ============================================================
                # COMMAND: TEST MATRIX
                # ============================================================
                if question.lower().startswith('test-matrix'):
                    parts = question.split(maxsplit=1)
                    if len(parts) < 2:
                        print("\n❌ Usage: test-matrix <questions_file>")
                        continue
                    
                    comparison = ConfigurationComparison(chatbot.config)
                    comparison.run_comparison(parts[1].strip(), test_mode="matrix")
                    continue
                
                # ============================================================
                # COMMAND: LOG
                # ============================================================
                if question.lower() == 'log':
                    print("\n📝 Session Logs:")
                    print(f"  Text: {chatbot.session_logger.get_log_path()}")
                    print(f"  JSON: {chatbot.session_logger.get_json_path()}")
                    continue
                
                # ============================================================
                # COMMAND: FORMATS
                # ============================================================
                if question.lower() == 'formats':
                    print("\n📑 Loaded Document Formats:")
                    for fmt_str, stats in chatbot.document_stats.items():
                        if stats['count'] > 0:
                            print(f"  {fmt_str}: {stats['success']}/{stats['count']} files loaded")
                            if stats['failed'] > 0:
                                print(f"    ({stats['failed']} failed)")
                    continue
                
                # ============================================================
                # COMMAND: STATS
                # ============================================================
                if question.lower() == 'stats':
                    stats = chatbot.session_logger._get_statistics()
                    if stats:
                        print("\n📊 Session Statistics:")
                        print(f"  Total queries: {stats.get('total_queries', 0)}")
                        print(f"  Avg response time: {stats.get('avg_response_time', 0):.2f}s")
                        print(f"  Fastest: {stats.get('min_response_time', 0):.2f}s")
                        print(f"  Slowest: {stats.get('max_response_time', 0):.2f}s")
                        print(f"  Avg sources used: {stats.get('avg_sources_used', 0):.1f}")
                    else:
                        print("\n📊 No queries yet in this session")
                    continue
                
                # ============================================================
                # REGULAR QUESTION PROCESSING
                # ============================================================
                print("\n⏳ Thinking...")
                response = chatbot.query(question)
                
                if "error" in response:
                    print(f"\n❌ Error: {response['error']}")
                    continue
                
                print(f"\n🤖 Bot: {response['answer']}")
                print(f"\n⏱️ Response time: {response['response_time']:.2f}s")
                
                if response.get('sources'):
                    print(f"\n📚 Sources ({len(response['sources'])} chunks used):")
                    print("=" * 70)
                    
                    for i, doc in enumerate(response['sources'][:5], 1):
                        filename = doc.metadata.get('filename', 'Unknown')
                        fmt = doc.metadata.get('format', 'unknown')
                        page = doc.metadata.get('page_number')
                        topic_type = doc.metadata.get('topic_type')
                        
                        # Source header
                        source_info = f"\n[{i}] {filename} [{fmt.upper()}]"
                        
                        if page:
                            source_info += f" (page {page})"
                        
                        if topic_type:
                            source_info += f" ({topic_type})"
                        
                        print(source_info)
                        print("-" * 70)
                    
                        # Display chunk text
                        chunk_text = doc.page_content
                        max_chunk_display = 300  # Show first 300 chars
                        
                        if len(chunk_text) <= max_chunk_display:
                            print(chunk_text)
                        else:
                            print(chunk_text[:max_chunk_display])
                            print(f"\n[... {len(chunk_text) - max_chunk_display} more characters ...]")
                        
                        print()
                    
                    if len(response['sources']) > 5:
                        print(f"... and {len(response['sources']) - 5} more sources")
                    
                    print("=" * 70)
            
            except KeyboardInterrupt:
                print("\n\n👋 Goodbye!")
                break
            except Exception as e:
                print(f"\n❌ Error: {e}")
                logger.exception("Error in interactive mode")
    
    finally:
        print("\n📝 Finalizing session log...")
        chatbot.finalize_session()
        print(f"✅ Session log saved: {chatbot.session_logger.get_log_path()}")
